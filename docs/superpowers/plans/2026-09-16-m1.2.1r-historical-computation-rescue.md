def _exact_object(value: Any, label: str, keys: set[str]) -> dict[str, Any]:
    current = _need_object(value, label)
    extras = set(current) - keys
    missing = keys - set(current)
    if extras:
        raise ArcaMechanicaBridgeError(f"{label} has unexpected fields: {sorted(extras)}")
    if missing:
        raise ArcaMechanicaBridgeError(f"{label} is missing fields: {sorted(missing)}")
    return current


def _content_digest(payload: dict[str, Any]) -> str:
    canonical = json.dumps(payload, ensure_ascii=False, sort_keys=True, separators=(",", ":"))
