def validate_reading(request: dict[str, Any]) -> dict[str, Any]:
    if request.get("format") != REQUEST_FORMAT:
        raise ArcaMechanicaBridgeError("unknown physical reading format")
    manifest = build_manifest()
    expected = manifest["critical_edition_carrier"]
    instance = _need_object(request.get("carrier_instance"), "carrier_instance")
    if instance.get("carrier_id") != expected["carrier_id"]:
        raise ArcaMechanicaBridgeError("unknown critical-edition carrier")
    if instance.get("location") != "workspace":
        raise ArcaMechanicaBridgeError("the carrier is not seated in the workspace")
    if instance.get("selected_band") != 1:
        raise ArcaMechanicaBridgeError("the selected band is untranscribed")
    if not isinstance(instance.get("instance_id"), str) or not instance["instance_id"].strip():
        raise ArcaMechanicaBridgeError("the physical carrier requires an instance id")
    tone = _need_object(request.get("tone"), "tone")
    if tone != {"number": 2, "witness": manifest["tone"]["witness"], "engaged": True}:
        raise ArcaMechanicaBridgeError("printed-p.51 Tone II must be engaged with its witness identity")
    return deepcopy(instance)


def execute_reading(request: dict[str, Any]) -> dict[str, Any]:
    instance = validate_reading(request)
    canonical = json.dumps(request, ensure_ascii=False, sort_keys=True, separators=(",", ":"))
    return {
        "format": BRIDGE_FORMAT,
        "request_fingerprint": hashlib.sha256(canonical.encode("utf-8")).hexdigest(),
        "reading": {
            "carrier_instance": instance,
            "selected_band": 1,
            "tone": deepcopy(request["tone"]),
            "classification": "H0 data through H1 critical-edition carrier",
        },
        "fragment": build_default_fragment(),
    }

export function isReadingReady(state: InstrumentState, manifest?: HistoricalManifest) {

```ts
export interface ReadingVoiceCell {
  degree: number
  pitchClass: string
  provenance?: HistoricalVoiceEvent['provenance']
}

export interface ReadingFrame {
  position: number
  eventNumber: number
  duration: { glyph: string; relativeMinimUnits: number; provenanceCell?: string }
  voices: Record<VoiceName, ReadingVoiceCell>
}

export function getReadingFrame(
  manifest: HistoricalManifest,
  fragment: HistoricalFragment | null,
  position: number,
): ReadingFrame {
  const index = Math.max(0, Math.min(5, Math.round(position)))
  const carrier = manifest.critical_edition_carrier
  const rows = carrier.pitch_source.content.rows
  const rhythm = carrier.rhythm_source.content
  const voices = Object.fromEntries(VOICE_NAMES.map((voice) => {
    const degree = rows[voice][index]
    const pitchClass = manifest.tone.degree_to_pitch_class[String(degree)]
    if (!pitchClass) throw new Error(`Tone-II mapping is missing degree ${degree}.`)
    return [voice, { degree, pitchClass, provenance: fragment?.events[index]?.voices[voice].provenance }]
  })) as Record<VoiceName, ReadingVoiceCell>
  return {
    position: index,
    eventNumber: index + 1,
    duration: {
      glyph: rhythm.glyphs[index],
      relativeMinimUnits: rhythm.relative_minim_units[index],
      provenanceCell: fragment?.events[index]?.rhythm_provenance.cell,
    },
    voices,
  }
}
```

Export a frozen `VOICE_NAMES` tuple and `getReadingFrames()` that maps positions
`0..5`. Validate equal row/rhythm lengths before indexing.
